- 客户端主动注册机制
- 心跳保活机制
- 在线状态管理
- 客户端统计监控
- 与现有日志系统的集成

## 🎯 实现目标

1. **客户端注册**：支持多种类型客户端主动注册
2. **心跳机制**：定期心跳保持在线状态
3. **状态管理**：实时跟踪客户端在线/离线状态
4. **统计监控**：提供详细的客户端统计信息

### 1. 客户端集成示例

#### JavaScript客户端
```javascript
class ClientManager {
    constructor(serverUrl, clientInfo) {
        this.serverUrl = serverUrl;
        this.clientInfo = clientInfo;
        this.clientId = null;
        this.heartbeatInterval = null;
    }
    
    async register() {
        const response = await fetch(`${this.serverUrl}/api/v1/clients/register`, {
            method: 'POST',
            headers: {'Content-Type': 'application/json'},
            body: JSON.stringify(this.clientInfo)
        });
        
        const result = await response.json();
        if (result.success) {
            this.clientId = result.client_id;
            this.startHeartbeat(result.heartbeat_interval);
        }
        return result;
    }
    
    startHeartbeat(interval) {
        this.heartbeatInterval = setInterval(async () => {
            await this.sendHeartbeat();
        }, interval * 1000);
    }
    
    async sendHeartbeat() {
        const response = await fetch(`${this.serverUrl}/api/v1/clients/heartbeat`, {
            method: 'POST',
            headers: {'Content-Type': 'application/json'},
            body: JSON.stringify({
                client_id: this.clientId,
                status: 'online'
            })
        });
        return response.json();
    }
}

// 使用示例
const client = new ClientManager('http://localhost:8080', {
    name: 'Web客户端',
    type: 'web',
    version: '1.0.0',
    description: 'Browser客户端'
});

client.register().then(result => {
    console.log('注册结果:', result);
});
```

#### Go客户端
```go
package main

import (
    "bytes"
    "encoding/json"
    "fmt"
    "net/http"
    "time"
)

type ClientSDK struct {
    ServerURL string
    ClientID  string
    client    *http.Client
}

func (c *ClientSDK) Register(req RegisterRequest) (*RegisterResponse, error) {
    data, _ := json.Marshal(req)
    resp, err := c.client.Post(
        c.ServerURL+"/api/v1/clients/register",
        "application/json",
        bytes.NewBuffer(data),
    )
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()
    
    var result RegisterResponse
    json.NewDecoder(resp.Body).Decode(&result)
    c.ClientID = result.ClientID
    
    return &result, nil
}

func (c *ClientSDK) StartHeartbeat(interval int) {
    ticker := time.NewTicker(time.Duration(interval) * time.Second)
    go func() {
        for range ticker.C {
            c.SendHeartbeat()
        }
    }()
}
```
